function fileStem(pathOrUrl) {
  const withoutQuery = pathOrUrl.split(/[?#]/)[0]
  const file = withoutQuery.split('/').pop() ?? ''
  const dot = file.lastIndexOf('.')
  return dot > 0 ? file.slice(0, dot) : file
}

  if (typeof value !== 'string') return null
